const SHEET_NAME      = "Data";                // nama sheet tab

// ============================================================
// doPost — menerima submission form pendaftaran
// ============================================================
function doPost(e) {
  try {
    var fileContent = e.parameter.fileContent;
    var filename    = e.parameter.filename;
    var email       = e.parameter.email;
    var nama        = e.parameter.nama;

    // Generate nomor referensi unik
    var refNum = "KBP-" + Utilities.formatDate(
      new Date(), Session.getScriptTimeZone(), "yyyyMMdd-HHmm"
    );

    // Upload file ke Google Drive
    var fileUrl = uploadFileToDrive(fileContent, filename);

    // Catat data ke Spreadsheet
    recordData(e, fileUrl, refNum);

    // Auto generate surat dari template → simpan di Drive → link ke Sheets
    generateSurat(e, refNum);

    // Kirim email konfirmasi ke pemohon
    kirimEmailKonfirmasi(e, refNum);

    return jsonResponse({ success: true, ref: refNum });

  } catch (err) {
    Logger.log("doPost error: " + err);
    return jsonResponse({ success: false, error: err.toString() });
  }
}

// ============================================================
// doGet — endpoint cek status oleh pemohon
// ============================================================
function doGet(e) {
  const action = e.parameter.action;

  if (action === "checkStatus") {
    return handleCheckStatus(e);
  }

  return jsonResponse({ error: "Action tidak dikenal" });
}

// ============================================================
// Upload file ke Google Drive
// ============================================================
function uploadFileToDrive(data, filename) {
  try {
    var folder = DriveApp.getFolderById(DRIVE_FOLDER_ID);

    var bytes = Utilities.base64Decode(data.split(',')[1]);
    var mime  = data.substring(5, data.indexOf(';'));
    var blob  = Utilities.newBlob(bytes, mime, filename);
    var file  = folder.createFile(blob);

    return file.getUrl();
  } catch (err) {
    Logger.log("uploadFileToDrive error: " + err);
    return "";
  }
}

// ============================================================
// Simpan data ke Google Sheets
// ============================================================
function recordData(e, fileUrl, refNum) {
  try {
    var ss    = SpreadsheetApp.openById(SPREADSHEET_ID);
    var sheet = ss.getSheetByName(SHEET_NAME);

    // Ambil header dari baris pertama
    var headers = sheet.getRange(1, 1, 1, sheet.getLastColumn())
                       .getValues()[0];

    var timestamp = Utilities.formatDate(
      new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss"
    );

    var row = headers.map(function(h) {
      h = h.toString().toLowerCase().trim();
      if (h === "timestamp")  return timestamp;
      if (h === "file")       return fileUrl;
      if (h === "status")     return "Pending";
      if (h === "ref_number") return refNum;
      if (h === "link_surat") return "";
      return e.parameter[h] || "";
    });

    sheet.getRange(sheet.getLastRow() + 1, 1, 1, row.length)
         .setValues([row]);

  } catch (err) {
    Logger.log("recordData error: " + err);
  }
}

// ============================================================
// Cek status oleh pemohon berdasarkan nomor referensi
// ============================================================
function handleCheckStatus(e) {
  try {
    const ref = (e.parameter.ref || "").trim().toUpperCase();
    if (!ref) return jsonResponse({ found: false });

    const ss      = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet   = ss.getSheetByName(SHEET_NAME);
    const data    = sheet.getDataRange().getValues();

    if (data.length < 2) return jsonResponse({ found: false });

    const headers = data[0].map(h => h.toString().toLowerCase().trim());
    const colRef  = headers.indexOf("ref_number");

    if (colRef === -1) {
      return jsonResponse({ found: false, error: "Kolom ref_number tidak ditemukan di sheet" });
    }

    for (let i = 1; i < data.length; i++) {
      const rowRef = (data[i][colRef] || "").toString().trim().toUpperCase();
      if (rowRef === ref) {
        const row = {};
        headers.forEach((h, idx) => {
          row[h] = data[i][idx] ? data[i][idx].toString() : "";
        });
        return jsonResponse({ found: true, row });
      }
    }

    return jsonResponse({ found: false });

  } catch (err) {
    return jsonResponse({ found: false, error: err.toString() });
  }
}

// Auto generate surat dari template Google Docs
// ============================================================
function generateSurat(e, refNum) {
  try {
    // Ambil data dari parameter form
    const nama            = e.parameter.nama            || "";
    const atasnama        = e.parameter.atasnama        || "";
    const fakultas        = e.parameter.fakultas        || "";
    const nama_lembaga    = e.parameter.nama_lembaga    || "";
    const nomor_surat     = e.parameter.nomor_surat     || "";
    const perihal         = e.parameter.perihal         || "";
    const alamat          = e.parameter.alamat          || "";
    const rt              = e.parameter.rukun_tetangga_rt || "";
    const rw              = e.parameter.rukun_warga_rw  || "";
    const kecamatan       = e.parameter.kecamatan       || "";
    const kabupaten       = e.parameter.kabupaten       || "";
    const pekerjaan       = e.parameter.pekerjaan       || "";
    const email           = e.parameter.email           || "";
    const nomor_hp        = e.parameter.nomor_hp        || "";
    const proposal        = e.parameter.proposal        || "";
    const lokasi          = e.parameter.lokasi          || "";
    const bidang          = e.parameter.bidang          || "";
    const tanggal_mulai   = e.parameter.tanggal_mulai   || "";
    const tanggal_selesai = e.parameter.tanggal_selesai || "";
    const penanggung_jawab= e.parameter.penanggung_jawab|| "";
    const anggota         = e.parameter.anggota_peneliti|| "";
    const tgl_surat       = e.parameter.tgl_surat       || "";

    // Salin template ke folder HASIL REGISTRASI
    const hasilFolder = DriveApp.getFolderById(HASIL_FOLDER_ID);
    const copy = DriveApp.getFileById(TEMPLATE_ID).makeCopy(
      `Surat Penelitian - ${nama} [${refNum}]`,
      hasilFolder
    );

    // Buka dokumen dan replace semua placeholder
    const doc  = DocumentApp.openById(copy.getId());
    const body = doc.getBody();

    body.replaceText("{{atasnama}}",         atasnama);
    body.replaceText("{{fakultas}}",          fakultas);
    body.replaceText("{{nama_lembaga}}",      nama_lembaga);
    body.replaceText("{{nomor_surat}}",       nomor_surat);
    body.replaceText("{{tgl_surat}}",         formatTanggalIndonesia(tgl_surat));
    body.replaceText("{{perihal}}",           perihal);
    body.replaceText("{{nama}}",              nama);
    body.replaceText("{{alamat}}",            alamat);
    body.replaceText("{{rukun_tetangga_rt}}", rt);
    body.replaceText("{{rukun_warga_rw}}",    rw);
    body.replaceText("{{kecamatan}}",         kecamatan);
    body.replaceText("{{kabupaten}}",         kabupaten);
    body.replaceText("{{pekerjaan}}",         pekerjaan);
    body.replaceText("{{email}}",             email);
    body.replaceText("{{nomor_hp}}",          nomor_hp);
    body.replaceText("{{proposal}}",          proposal);
    body.replaceText("{{lokasi}}",            lokasi);
    body.replaceText("{{bidang}}",            bidang);
    body.replaceText("{{tanggal_mulai}}",     formatTanggalIndonesia(tanggal_mulai));
    body.replaceText("{{tanggal_selesai}}",   formatTanggalIndonesia(tanggal_selesai));
    body.replaceText("{{penanggung_jawab}}",  penanggung_jawab);
    body.replaceText("{{anggota_peneliti}}",  anggota);
    body.replaceText("{{ref_number}}",        refNum);

    doc.saveAndClose();

    // Simpan link surat ke kolom link_surat di Sheets
    const ss      = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet   = ss.getSheetByName(SHEET_NAME);
    const data    = sheet.getDataRange().getValues();
    const headers = data[0].map(h => h.toString().toLowerCase().trim());
    const colRef  = headers.indexOf("ref_number") + 1;
    const colLink = headers.indexOf("link_surat") + 1;

    if (colRef && colLink) {
      for (let i = 1; i < data.length; i++) {
        if ((data[i][colRef - 1] || "").toString().trim() === refNum) {
          sheet.getRange(i + 1, colLink).setValue(copy.getUrl());

          // Auto update status → Diproses
          const colStatus = headers.indexOf("status") + 1;
          if (colStatus) {
            sheet.getRange(i + 1, colStatus).setValue("Diproses");
          }

          break;
        }
      }
    }

    Logger.log("Surat berhasil dibuat: " + copy.getUrl());
    return copy.getUrl();

  } catch (err) {
    Logger.log("generateSurat error: " + err);
    return null;
  }
}

function formatTanggalIndonesia(dateStr) {
  if (!dateStr) return "-";
  const bulan = [
    "Januari","Februari","Maret","April","Mei","Juni",
    "Juli","Agustus","September","Oktober","November","Desember"
  ];
  const d = new Date(dateStr);
  if (isNaN(d)) return dateStr;
  return `${d.getDate()} ${bulan[d.getMonth()]} ${d.getFullYear()}`;
}

    .setMimeType(ContentService.MimeType.JSON);
}

// SETUP SHEET — jalankan SEKALI manual dari GAS editor
// Fungsi ini: buat header + dropdown status otomatis
// Cara: buka GAS editor → pilih fungsi setupSheet → klik Run
// ============================================================
function setupSheet() {
  const ss    = SpreadsheetApp.openById(SPREADSHEET_ID);
  let sheet   = ss.getSheetByName(SHEET_NAME);

  // Buat sheet "Data" kalau belum ada
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
  }

  // Set header baris 1 kalau masih kosong
  const headers = [
    "timestamp", "atasnama", "fakultas", "nama_lembaga",
    "nomor_surat", "tgl_surat", "perihal", "nama", "alamat",
    "rukun_tetangga_rt", "rukun_warga_rw", "kecamatan", "kabupaten",
    "pekerjaan", "proposal", "lokasi", "bidang",
    "tanggal_mulai", "tanggal_selesai", "penanggung_jawab",
    "anggota_peneliti", "email", "nomor_hp",
    "file", "link_surat", "status", "ref_number"
  ];

  const firstRow = sheet.getRange(1, 1, 1, headers.length).getValues()[0];
  const isEmpty  = firstRow.every(cell => cell === "");
  if (isEmpty) {
    sheet.getRange(1, 1, 1, headers.length).setValues([headers]);

    // Bold + warna header
    const headerRange = sheet.getRange(1, 1, 1, headers.length);
    headerRange.setFontWeight("bold")
               .setBackground("#0a2a5e")
               .setFontColor("#ffffff")
               .setHorizontalAlignment("center");

    // Freeze baris header
    sheet.setFrozenRows(1);

    Logger.log("Header berhasil dibuat.");
  }

  // Temukan kolom "status"
  const allHeaders = sheet.getRange(1, 1, 1, sheet.getLastColumn())
                          .getValues()[0]
                          .map(h => h.toString().toLowerCase().trim());
  const colStatus  = allHeaders.indexOf("status") + 1;

  if (!colStatus) {
    Logger.log("Kolom status tidak ditemukan.");
    return;
  }

  // Pasang dropdown validasi di kolom status (baris 2 ke bawah, 1000 baris)
  const statusRange = sheet.getRange(2, colStatus, 1000, 1);
  const rule = SpreadsheetApp.newDataValidation()
    .requireValueInList(["Pending", "Diproses", "Selesai", "Ditolak"], true)
    .setAllowInvalid(false)
    .setHelpText("Pilih status: Pending, Diproses, Selesai, atau Ditolak")
    .build();

  statusRange.setDataValidation(rule);

  // Kasih warna kondisional per status (conditional formatting)
  const rules = sheet.getConditionalFormatRules();

  const colorMap = [
    { val: "Pending",   bg: "#fef3c7", font: "#92400e" },
    { val: "Diproses",  bg: "#dbeafe", font: "#1e40af" },
    { val: "Selesai",   bg: "#d1fae5", font: "#065f46" },
    { val: "Ditolak",   bg: "#fee2e2", font: "#991b1b" },
  ];

  colorMap.forEach(({ val, bg, font }) => {
    const newRule = SpreadsheetApp.newConditionalFormatRule()
      .whenTextEqualTo(val)
      .setBackground(bg)
      .setFontColor(font)
      .setBold(true)
      .setRanges([statusRange])
      .build();
    rules.push(newRule);
  });

  sheet.setConditionalFormatRules(rules);

  Logger.log("✅ Setup selesai! Dropdown + warna status sudah terpasang.");
